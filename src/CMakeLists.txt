function(NavalCommandAddDomainTarget domain)
    set(target "${domain}.Api")
    add_library("${target}" INTERFACE)
    target_include_directories("${target}" INTERFACE "${PROJECT_SOURCE_DIR}/public")
    if(ARGN)
        target_link_libraries("${target}" INTERFACE ${ARGN})
    endif()
    NavalCommandConfigureDomainTarget("${target}")
endfunction()
