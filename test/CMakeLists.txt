# Compile each public header as the first and only domain include in its own translation unit.
file(GLOB_RECURSE domainHeaders CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/public/*.h")
foreach(header IN LISTS domainHeaders)
    file(RELATIVE_PATH relativeHeader "${PROJECT_SOURCE_DIR}/public" "${header}")
    string(MAKE_C_IDENTIFIER "${relativeHeader}" headerTarget)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${headerTarget}.cpp")
    file(WRITE "${source}" "#include <${relativeHeader}>\n")
    add_library("header-${headerTarget}" OBJECT "${source}")
    target_link_libraries("header-${headerTarget}" PRIVATE World.Api)
    NavalCommandConfigureTarget("header-${headerTarget}")
endforeach()
